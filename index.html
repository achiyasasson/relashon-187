הפעלת התראות באייפון (iPhone - iOS 16.4+)

דורש הוספה למסך הבית

מדיניות האבטחה של אפל מאפשרת קבלת התראות Push באייפון רק לאתרים שנוספו למסך הבית:

  1. פתחו את האתר בדפדפן Safari במכשיר ה-iPhone.
  2. לחצו על כפתור השיתוף (ריבוע עם חץ כלפי מעלה בתחתית המסך).
  3. גללו מטה ובחרו "הוסף למסך הבית" (Add to Home Screen ➕).
  4. פתחו את האפליקציה מהאייקון שנוצר במסך הבית, לחצו על הפעמון 🔔 ואשרו התראות.

למה כדאי להוסיף את המערכת למסך הבית?

✨ מסך מלא (Full Screen) פועל בדיוק כמו אפליקציה צבאית אמיתית, ללא שורת הכתובת של הדפדפן.
⚡ גישה מיידית לחיצה אחת מהאייקון במסך הבית ללא צורך בהקלדת קישור או חיפוש בהיסטוריה.
🔔 התראות רקע מאפשר קבלת התראות מלאה גם באייפון ובאנדרואיד.
הוראות באייפון (Safari):

1. לחצו על כפתור השיתוף (אייקון הריבוע עם החץ למעלה ⬆️ באמצע השורה התחתונה).
2. בתפריט שנפתח, גללו למטה ולחצו על "הוסף למסך הבית" (Add to Home Screen).
3. לחצו על "הוסף" (Add) בפינה העליונה.

הוראות באנדרואיד (Chrome):

1. לחצו על כפתור 3 הנקודות (⋮) בפינה העליונה של הדפדפן.
2. לחצו על "התקן אפליקציה" (Install app) או "הוסף למסך דף הבית".
3. אשרו את ההוספה והאפליקציה תופיע במסך הבית שלכם.

// === Google Apps Script: שרת סנכרון יומן 187 ומערכת רלש״ון ===
function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) || 'get_events';
  
  if (action === 'get_events' || action === 'getEvents') {
    try {
      var cal = CalendarApp.getCalendarById('187brigade@gmail.com') || CalendarApp.getDefaultCalendar();
      var now = new Date();
      var start = new Date(now.getFullYear(), now.getMonth() - 2, 1);
      var end = new Date(now.getFullYear(), now.getMonth() + 4, 1);
      var events = cal.getEvents(start, end);
      
      var result = events.map(function(ev) {
        var summary = ev.getTitle();
        var cat = 'meeting';
        if (summary.indexOf('אלת') !== -1 || summary.indexOf('תרגד') !== -1 || summary.indexOf('אימון') !== -1) cat = 'training';
        else if (summary.indexOf('תעמ') !== -1 || summary.indexOf('תע\"מ') !== -1 || summary.indexOf('מבצעי') !== -1) cat = 'operational';
        else if (summary.indexOf('טקס') !== -1 || summary.indexOf('עצרת') !== -1 || summary.indexOf('נופש') !== -1 || summary.indexOf('חנוכת') !== -1) cat = 'ceremony';
        
        return {
          id: ev.getId(),
          summary: summary,
          description: ev.getDescription() || '',
          location: ev.getLocation() || '',
          start: ev.isAllDayEvent() ? Utilities.formatDate(ev.getAllDayStartDate(), 'Asia/Jerusalem', 'yyyy-MM-dd') : ev.getStartTime().toISOString(),
          end: ev.isAllDayEvent() ? Utilities.formatDate(ev.getAllDayEndDate(), 'Asia/Jerusalem', 'yyyy-MM-dd') : ev.getEndTime().toISOString(),
          allDay: ev.isAllDayEvent(),
          category: cat,
          htmlLink: 'https://calendar.google.com/calendar/u/0/r?cid=187brigade@gmail.com',
          attendees: ev.getGuestList().map(function(g) {
            return {
              email: g.getEmail(),
              displayName: g.getName() || g.getEmail(),
              responseStatus: g.getGuestStatus().toString().toLowerCase()
            };
          })
        };
      });
      
      return ContentService.createTextOutput(JSON.stringify({status: 'success', events: result}))
        .setMimeType(ContentService.MimeType.JSON);
    } catch(err) {
      return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()}))
        .setMimeType(ContentService.MimeType.JSON);
    }
  }
  
  var props = PropertiesService.getScriptProperties();
  var saved = props.getProperty('RELASHON_DATA');
  return ContentService.createTextOutput(saved || JSON.stringify({empty: true}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var raw = e.postData.contents;
    var data = JSON.parse(raw);
    
    if (data.action === 'create_event') {
      var cal = CalendarApp.getCalendarById('187brigade@gmail.com') || CalendarApp.getDefaultCalendar();
      var ev;
      if (data.allDay) {
        ev = cal.createAllDayEvent(data.summary, new Date(data.start), {
          description: data.description || '',
          location: data.location || ''
        });
      } else {
        ev = cal.createEvent(data.summary, new Date(data.start), new Date(data.end), {
          description: data.description || '',
          location: data.location || '',
          guests: (data.attendeeEmails || []).join(',')
        });
      }
      return ContentService.createTextOutput(JSON.stringify({status: 'success', id: ev.getId()}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    PropertiesService.getScriptProperties().setProperty('RELASHON_DATA', raw);
    return ContentService.createTextOutput(JSON.stringify({status: 'saved'}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
